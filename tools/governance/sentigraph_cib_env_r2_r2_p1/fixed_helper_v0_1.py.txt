SAFE_IDENTIFIER = re.compile(r"^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$")
SAFE_JSON_BASENAME = re.compile(r"^[A-Za-z0-9][A-Za-z0-9._-]{0,154}\.json$")
PROVIDER_PACKAGE_FIELDS = (
    "package_root_ref",
    "package_relative_path",
    "package_index_ref",
    "package_role",
    "package_id",
)
SAFE_METADATA_KEYS = frozenset(
    {
        "package_id",
        "package_name",
        "package_role",
        "package_index_ref",
        "package_root_ref",
        "package_relative_path",

        "validation_status",
    }
)
MARKER_FILENAMES = (
    "manifest.json",
    "validation_report.json",
    "coverage_note.md",
    "README.md",
    "validation_report.md",
)
MAX_CANDIDATES = 16
MAX_SAFE_METADATA_READS = 48


class Blocked(Exception):
    pass


class BroadcastBlocked(Exception):
    pass


def _reject_duplicate_keys(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise Blocked()
        result[key] = value
    return result


def _read_strict_json_object(path):
    try:
        with open(path, "rb") as handle:
            raw = handle.read()
    except OSError as exc:
        raise Blocked() from exc
    if raw.startswith(b"\xef\xbb\xbf"):
        raise Blocked()
    try:
        text = raw.decode("utf-8", "strict")
        value = json.loads(text, object_pairs_hook=_reject_duplicate_keys)
    except (UnicodeDecodeError, json.JSONDecodeError, Blocked) as exc:
        raise Blocked() from exc
    if not isinstance(value, dict):
        raise Blocked()
    return value


def _is_safe_identifier(value):
    return isinstance(value, str) and SAFE_IDENTIFIER.fullmatch(value) is not None


def _is_safe_json_basename(value):
    if not isinstance(value, str):
        return False
    if "/" in value or "\\" in value or value in {".", ".."}:
        return False
    return SAFE_JSON_BASENAME.fullmatch(value) is not None


def _safe_posix_relative_parts(value):
    if not isinstance(value, str):
        raise Blocked()
    if not value or value.startswith("/") or "\\" in value:
        raise Blocked()
    parts = value.split("/")
    if not parts or any(part in {"", ".", ".."} for part in parts):
        raise Blocked()
    if any(not _is_safe_identifier(part) for part in parts):
        raise Blocked()
    return tuple(parts)


def _is_reparse(file_stat):
    return bool(
        getattr(file_stat, "st_file_attributes", 0)
        & stat.FILE_ATTRIBUTE_REPARSE_POINT
    )


def _walk_tree(root, depth=0):
    if depth > MAX_SEARCH_DEPTH:
        return
    try:
        entries = list(os.scandir(root))
    except OSError as exc:
        raise Blocked() from exc
    entries.sort(key=lambda entry: entry.name)
    for entry in entries:
        try:
            if entry.is_symlink():
                continue
            file_stat = entry.stat(follow_symlinks=False)
            if getattr(file_stat, "st_file_attributes", 0) & stat.FILE_ATTRIBUTE_REPARSE_POINT:
                continue
            if entry.is_dir(follow_symlinks=False):
                yield entry.path, True
                if depth < MAX_SEARCH_DEPTH:
                    yield from _walk_tree(entry.path, depth + 1)
            elif entry.is_file(follow_symlinks=False):
                yield entry.path, False
        except OSError as exc:
            raise Blocked() from exc


def _find_exact_provider_result():
    match = None
    for path, is_directory in _walk_tree(SEARCH_ROOT):
        if is_directory:
            continue
        if os.path.basename(path) != PROVIDER_RESULT_BASENAME:
            continue
        if match is not None:
            raise Blocked()
        match = path
    if match is None:
        raise Blocked()
    return match


def _provider_contract(provider):
    for field_name in PROVIDER_PACKAGE_FIELDS:
        if field_name not in provider:
            raise Blocked()
    package_root_ref = provider["package_root_ref"]
    package_relative_path = provider["package_relative_path"]
    package_index_ref = provider["package_index_ref"]
    package_role = provider["package_role"]
    package_id = provider["package_id"]
    if package_root_ref != "configured_export_root":
        raise Blocked()
    relative_parts = _safe_posix_relative_parts(package_relative_path)
    if not _is_safe_json_basename(package_index_ref):
        raise Blocked()
    if not _is_safe_identifier(package_role):
        raise Blocked()
    if not _is_safe_identifier(package_id):
        raise Blocked()
    if relative_parts[-1] != package_id:
        raise Blocked()
    return (
        package_relative_path,
        relative_parts,
        package_index_ref,
        package_role,
        package_id,
    )


def _regular_nonreparse_file(path):
    try:
        file_stat = os.stat(path, follow_symlinks=False)
    except OSError:
        return False
    if _is_reparse(file_stat):
        return False
    return stat.S_ISREG(file_stat.st_mode)


def _lexical_candidate_export_root(candidate_path, relative_parts):
    lexical_tail = os.sep.join(relative_parts)
    suffix = os.sep + lexical_tail
    if not candidate_path.endswith(suffix):
        raise Blocked()
    export_root = candidate_path[: -len(suffix)]
    if not export_root:
        raise Blocked()
    return export_root


def _iter_safe_metadata_values(value):
    if isinstance(value, dict):
        for key, child in value.items():
            if key in SAFE_METADATA_KEYS:
                yield key, child
            yield from _iter_safe_metadata_values(child)
    elif isinstance(value, list):
        for child in value:
            yield from _iter_safe_metadata_values(child)


def _metadata_matches(
    metadata_objects,
    package_relative_path,
    package_index_ref,
    package_role,
    package_id,
):
    saw_id_or_name = False
    saw_role = False
    for metadata in metadata_objects:
        for key, value in _iter_safe_metadata_values(metadata):
            if key in {"package_id", "package_name"}:
                if value != package_id:
                    return False
                saw_id_or_name = True
            elif key == "package_role":
                if value != package_role:
                    return False
                saw_role = True
            elif key == "package_index_ref":
                if value != package_index_ref:
                    return False
            elif key == "package_root_ref":
                if value != "configured_export_root":
                    return False
            elif key == "package_relative_path":
                if value != package_relative_path:
                    return False
    return saw_id_or_name and saw_role


def _find_unique_package_candidate(
    package_relative_path,
    relative_parts,
    package_index_ref,
    package_role,
    package_id,
):
    candidate_count = 0
    conforming = None
    safe_metadata_reads = 0
    for path, is_directory in _walk_tree(SEARCH_ROOT):
        if not is_directory or os.path.basename(path) != package_id:
            continue
        candidate_count += 1
        if candidate_count > MAX_CANDIDATES:
            raise Blocked()
        try:
            export_root = _lexical_candidate_export_root(path, relative_parts)
        except Blocked:
            continue
        marker_names = MARKER_FILENAMES + (package_index_ref,)
        if any(
            not _regular_nonreparse_file(os.path.join(path, marker_name))
            for marker_name in marker_names
        ):
            continue
        metadata_objects = []
        distinct_paths = []
        for metadata_name in (package_index_ref, "manifest.json", "validation_report.json"):
            metadata_path = os.path.join(path, metadata_name)
            if metadata_path not in distinct_paths:
                distinct_paths.append(metadata_path)
        if safe_metadata_reads + len(distinct_paths) > MAX_SAFE_METADATA_READS:
            raise Blocked()
        for metadata_path in distinct_paths:
            metadata_objects.append(_read_strict_json_object(metadata_path))
            safe_metadata_reads += 1
        if not _metadata_matches(
            metadata_objects,
            package_relative_path,
            package_index_ref,
            package_role,
            package_id,
        ):
            continue
        if conforming is not None:
            raise Blocked()
        conforming = (path, export_root)
    if conforming is None:
        raise Blocked()
    return conforming


def _validated_configuration_value(value):
    if not isinstance(value, str):
        raise Blocked()
    if not 1 <= len(value) <= 2048:
        raise Blocked()
    if value != value.strip():
        raise Blocked()
    if "\x00" in value or not value.isprintable():
        raise Blocked()
    return value


def _derive_configuration(provider_path, provider, export_root):
    adapter_id = provider.get("adapter_id")
    if not _is_safe_identifier(adapter_id):
        raise Blocked()
    results_dir = os.path.dirname(provider_path)
    values = (
        _validated_configuration_value(results_dir),
        _validated_configuration_value(export_root),
        _validated_configuration_value(adapter_id),
    )
    return values


def _broadcast_environment_change():
    try:
        user32 = ctypes.WinDLL("user32", use_last_error=True)
        send_message_timeout_w = user32.SendMessageTimeoutW
        send_message_timeout_w.argtypes = (
            ctypes.wintypes.HWND,
            ctypes.wintypes.UINT,
            ctypes.wintypes.WPARAM,
            ctypes.wintypes.LPARAM,
            ctypes.wintypes.UINT,
            ctypes.wintypes.UINT,
            ctypes.POINTER(ctypes.c_size_t),
        )
        send_message_timeout_w.restype = ctypes.wintypes.LPARAM
        environment_text = ctypes.c_wchar_p(ENVIRONMENT_BROADCAST_LPARAM)
        lparam_value = ctypes.cast(environment_text, ctypes.c_void_p).value
        if lparam_value is None:
            raise BroadcastBlocked()
        message_result = ctypes.c_size_t(0)
        ctypes.set_last_error(0)
        broadcast_result = send_message_timeout_w(
            HWND_BROADCAST,
            WM_SETTINGCHANGE,
            0,
            lparam_value,
            SMTO_ABORTIFHUNG,
            BROADCAST_TIMEOUT_MS,
            ctypes.byref(message_result),
        )
        if broadcast_result == 0:
            raise BroadcastBlocked()
    except BroadcastBlocked:
        raise
    except Exception as exc:
        raise BroadcastBlocked() from exc


def _write_and_readback_current_user_environment(values):
    access = winreg.KEY_SET_VALUE | winreg.KEY_QUERY_VALUE
    try:
        with winreg.OpenKey(
            winreg.HKEY_CURRENT_USER,
            REGISTRY_SUBKEY,
            0,
            access,
        ) as key_handle:
            for variable_name, value in zip(ENVIRONMENT_VARIABLE_NAMES, values):
                winreg.SetValueEx(key_handle, variable_name, 0, winreg.REG_SZ, value)
            for variable_name, expected_value in zip(ENVIRONMENT_VARIABLE_NAMES, values):
                actual_value, actual_type = winreg.QueryValueEx(key_handle, variable_name)
                if actual_type != winreg.REG_SZ or actual_value != expected_value:
                    raise Blocked()
    except OSError as exc:
        raise Blocked() from exc
    _broadcast_environment_change()


def _run():
    provider_path = _find_exact_provider_result()
    provider = _read_strict_json_object(provider_path)
    (
        package_relative_path,
        relative_parts,
        package_index_ref,
        package_role,
        package_id,
    ) = _provider_contract(provider)
    _, export_root = _find_unique_package_candidate(
        package_relative_path,
        relative_parts,
        package_index_ref,
        package_role,
        package_id,
    )
    values = _derive_configuration(provider_path, provider, export_root)
    _write_and_readback_current_user_environment(values)
    print(OUTPUT_BROADCAST_PASS)
    print(OUTPUT_RESULTS_DIR_PASS)
    print(OUTPUT_EXPORT_ROOT_PASS)
    print(OUTPUT_ADAPTER_ID_PASS)
    print(OUTPUT_REPAIR_PASS)


def main():
    try:
        _run()
    except BroadcastBlocked:
        print(OUTPUT_BROADCAST_BLOCKED)
        return 1
    except Exception:
        print(OUTPUT_BLOCKED)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
